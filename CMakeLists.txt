project(TrackRS VERSION 0.6.7 LANGUAGES CXX)
include(CTest)
set(TRACKRS_SANITIZER "" CACHE STRING "address,undefined or thread (separate builds)")
configure_file(cmake/version.h.in generated/trackrs_version.h @ONLY)
configure_file(cmake/version.svg.in generated/version.svg @ONLY)
file(READ "${CMAKE_SOURCE_DIR}/data/metainfo/org.trackrs.TrackRS.metainfo.xml" TRACKRS_APPDATA)
string(REPLACE "</component>" "<custom><value key=\"TrackRS::Version\">${PROJECT_VERSION}</value></custom>\n</component>" TRACKRS_APPDATA "${TRACKRS_APPDATA}")
file(WRITE "${CMAKE_BINARY_DIR}/generated/org.trackrs.TrackRS.metainfo.xml" "${TRACKRS_APPDATA}")
option(TRACKRS_LTO "Enable supported release link-time optimization" ON)
include(CheckIPOSupported)
if(TRACKRS_LTO AND NOT TRACKRS_SANITIZER)
  check_ipo_supported(RESULT TRACKRS_IPO_SUPPORTED OUTPUT TRACKRS_IPO_REASON)
endif()
option(TRACKRS_NATIVE "Optimize for the build machine (not portable)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_path(glm_INCLUDE_DIR glm/glm.hpp REQUIRED)
find_path(json_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
list(APPEND TRACKRS_INCLUDES "${glm_INCLUDE_DIR}" "${json_INCLUDE_DIR}")
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
foreach(dep IN ITEMS SDL2 SDL2_image GLEW physfs openal alut tinyxml2)
  if(dep STREQUAL "GLEW")
    set(header GL/glew.h)
  elseif(dep STREQUAL "openal")
    set(header AL/al.h)
  elseif(dep STREQUAL "alut")
    set(header AL/alut.h)
  elseif(dep STREQUAL "SDL2_image")
    set(header SDL2/SDL_image.h)
  elseif(dep STREQUAL "SDL2")
    set(header SDL2/SDL.h)
  else()
    set(header "${dep}.h")
  endif()
  find_path(${dep}_INCLUDE_DIR NAMES "${header}" REQUIRED)
  set(names "${dep}")
  if(dep STREQUAL "GLEW")
    list(APPEND names glew32 glew32s)
  elseif(dep STREQUAL "openal")
    list(APPEND names OpenAL32)
  endif()
  find_library(${dep}_LIBRARY NAMES ${names} REQUIRED)
  list(APPEND TRACKRS_INCLUDES "${${dep}_INCLUDE_DIR}")
  list(APPEND TRACKRS_LIBRARIES "${${dep}_LIBRARY}")
endforeach()
# Engine components compile independently; the application composes them.
file(GLOB TRACKRS_CORE CONFIGURE_DEPENDS src/PEngine/util.cpp src/PEngine/vmath.cpp src/PEngine/physfs_rw.cpp src/PEngine/validation.cpp)
file(GLOB TRACKRS_SIM CONFIGURE_DEPENDS src/PSim/*.cpp)
file(GLOB TRACKRS_GAME CONFIGURE_DEPENDS src/Trigger/*.cpp)
add_library(trackrs_core OBJECT ${TRACKRS_CORE})
add_library(trackrs_sim OBJECT ${TRACKRS_SIM})
add_library(trackrs_world OBJECT src/PEngine/terrain.cpp src/PEngine/rigidity.cpp)
add_library(trackrs_audio OBJECT src/PEngine/audio.cpp)
add_library(trackrs_ui OBJECT src/PEngine/ui.cpp)
add_library(trackrs_platform OBJECT src/PEngine/app.cpp src/PEngine/config.cpp)
add_library(trackrs_renderer OBJECT src/PEngine/render.cpp src/PEngine/texture.cpp src/PEngine/model.cpp src/PEngine/fxman.cpp src/PEngine/vbuffer.cpp engine/renderer/renderer.cpp)
add_executable(trackrs ${TRACKRS_GAME})
set(TRACKRS_COMPONENTS trackrs_core trackrs_sim trackrs_world trackrs_audio trackrs_ui trackrs_platform trackrs_renderer)
target_link_libraries(trackrs PRIVATE ${TRACKRS_COMPONENTS})
set_property(SOURCE src/Trigger/main.cpp APPEND PROPERTY COMPILE_DEFINITIONS
  TRACKRS_BUILD_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
foreach(component IN ITEMS trackrs ${TRACKRS_COMPONENTS})
target_include_directories(${component} PRIVATE "${CMAKE_BINARY_DIR}/generated" src/include ${TRACKRS_INCLUDES})
target_link_libraries(${component} PRIVATE ${TRACKRS_LIBRARIES} OpenGL::GL Threads::Threads)
target_compile_definitions(${component} PRIVATE SDL_MAIN_HANDLED PACKAGE_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_compile_definitions(${component} PRIVATE WIN32 NOMINMAX)
else()
  target_compile_definitions(${component} PRIVATE UNIX)
endif()
if(MSVC)
  # FreeALUT's MSVC header uses <al.h>/<alc.h>, while legacy callers use <AL/...>.
  target_include_directories(${component} PRIVATE "${openal_INCLUDE_DIR}/AL")
  target_compile_options(${component} PRIVATE /W4 /permissive- /fp:strict)
else()
  target_compile_options(${component} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -ffp-contract=off)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${component} PRIVATE -Werror=mismatched-tags)
  endif()
  if(TRACKRS_NATIVE)
    target_compile_options(${component} PRIVATE -march=native)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    target_compile_options(${component} PRIVATE -march=x86-64-v2)
  endif()
  if(TRACKRS_SANITIZER)
    target_compile_options(${component} PRIVATE "-fsanitize=${TRACKRS_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(${component} PRIVATE "-fsanitize=${TRACKRS_SANITIZER}")
  endif()
endif()
if(TRACKRS_IPO_SUPPORTED)
  set_property(TARGET ${component} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
endforeach()
