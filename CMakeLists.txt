cmake_minimum_required(VERSION 3.20)
project(TrackRS VERSION 0.6.7 LANGUAGES CXX)
include(CTest)
option(TRACKRS_NATIVE "Optimize for the build machine (not portable)" OFF)
set(TRACKRS_SANITIZER "" CACHE STRING "address,undefined or thread (separate builds)")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
foreach(dep IN ITEMS SDL2 SDL2_image GLEW physfs openal alut tinyxml2)
  if(dep STREQUAL "GLEW")
    set(header GL/glew.h)
  elseif(dep STREQUAL "openal")
    set(header AL/al.h)
  elseif(dep STREQUAL "alut")
    set(header AL/alut.h)
  elseif(dep STREQUAL "SDL2_image")
    set(header SDL2/SDL_image.h)
  elseif(dep STREQUAL "SDL2")
    set(header SDL2/SDL.h)
  else()
    set(header "${dep}.h")
  endif()
  find_path(${dep}_INCLUDE_DIR NAMES "${header}" REQUIRED)
  set(names "${dep}")
  if(dep STREQUAL "GLEW")
    list(APPEND names glew32 glew32s)
  elseif(dep STREQUAL "openal")
    list(APPEND names OpenAL32)
  endif()
  find_library(${dep}_LIBRARY NAMES ${names} REQUIRED)
  list(APPEND TRACKRS_INCLUDES "${${dep}_INCLUDE_DIR}")
  list(APPEND TRACKRS_LIBRARIES "${${dep}_LIBRARY}")
endforeach()
file(GLOB_RECURSE TRACKRS_SOURCES CONFIGURE_DEPENDS src/PEngine/*.cpp src/PSim/*.cpp src/Trigger/*.cpp)
add_executable(trackrs ${TRACKRS_SOURCES})
set_property(SOURCE src/Trigger/main.cpp APPEND PROPERTY COMPILE_DEFINITIONS
  TRACKRS_BUILD_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
target_include_directories(trackrs PRIVATE src/include ${TRACKRS_INCLUDES})
target_link_libraries(trackrs PRIVATE ${TRACKRS_LIBRARIES} OpenGL::GL Threads::Threads)
target_compile_definitions(trackrs PRIVATE SDL_MAIN_HANDLED PACKAGE_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_compile_definitions(trackrs PRIVATE WIN32 NOMINMAX)
else()
  target_compile_definitions(trackrs PRIVATE UNIX)
endif()
if(MSVC)
  target_compile_options(trackrs PRIVATE /W4 /permissive- /fp:strict)
else()
  target_compile_options(trackrs PRIVATE -Wall -Wextra -Wpedantic -Wconversion -ffp-contract=off)
  if(TRACKRS_NATIVE)
    target_compile_options(trackrs PRIVATE -march=native)
  endif()
  if(TRACKRS_SANITIZER)
    target_compile_options(trackrs PRIVATE "-fsanitize=${TRACKRS_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(trackrs PRIVATE "-fsanitize=${TRACKRS_SANITIZER}")
  endif()
endif()
# Build-tree launch works from any directory; installed data layout is unchanged.
file(READ "${CMAKE_SOURCE_DIR}/bin/trackrs.config.defs" TRACKRS_CONFIG)
string(REPLACE "../data" "${CMAKE_SOURCE_DIR}/data" TRACKRS_CONFIG "${TRACKRS_CONFIG}")
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:trackrs>/trackrs.config.defs" CONTENT "${TRACKRS_CONFIG}")
if(BUILD_TESTING)
  add_executable(validation_tests tests/validation_tests.cpp src/PEngine/validation.cpp)
  target_include_directories(validation_tests PRIVATE src/include tests/vendor)
  if(NOT MSVC AND TRACKRS_SANITIZER)
    target_compile_options(validation_tests PRIVATE "-fsanitize=${TRACKRS_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(validation_tests PRIVATE "-fsanitize=${TRACKRS_SANITIZER}")
  endif()
  add_test(NAME validation-unit COMMAND validation_tests)
  add_test(NAME cli-help COMMAND trackrs --help)
  add_test(NAME cli-invalid COMMAND trackrs --headless --frames 0)
  set_tests_properties(cli-invalid PROPERTIES WILL_FAIL TRUE)
endif()
