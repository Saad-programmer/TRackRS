  elseif(dep STREQUAL "alut")
    set(header AL/alut.h)
  elseif(dep STREQUAL "SDL2_image")
    set(header SDL2/SDL_image.h)
  elseif(dep STREQUAL "SDL2")
    set(header SDL2/SDL.h)
  else()
    set(header "${dep}.h")
  endif()
  find_path(${dep}_INCLUDE_DIR NAMES "${header}" REQUIRED)
  set(names "${dep}")
  if(dep STREQUAL "openal")
    list(APPEND names OpenAL32)
  endif()
  find_library(${dep}_LIBRARY NAMES ${names} REQUIRED)
  list(APPEND TRACKRS_INCLUDES "${${dep}_INCLUDE_DIR}")
  list(APPEND TRACKRS_LIBRARIES "${${dep}_LIBRARY}")
endforeach()

  # FreeALUT's MSVC header uses <al.h>/<alc.h>, while legacy callers use <AL/...>.
  target_include_directories(${component} PRIVATE "${openal_INCLUDE_DIR}/AL")
  target_compile_options(${component} PRIVATE /W4 /permissive- /fp:strict)
else()
  target_compile_options(${component} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -ffp-contract=off)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${component} PRIVATE -Werror=mismatched-tags)
  endif()
  if(TRACKRS_NATIVE)
    target_compile_options(${component} PRIVATE -march=native)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    target_compile_options(${component} PRIVATE -march=x86-64-v2)
  endif()
  if(TRACKRS_SANITIZER)
    target_compile_options(${component} PRIVATE "-fsanitize=${TRACKRS_SANITIZER}" -fno-omit-frame-pointer)
    target_link_options(${component} PRIVATE "-fsanitize=${TRACKRS_SANITIZER}")
  endif()
endif()

  set_property(TARGET ${component} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
